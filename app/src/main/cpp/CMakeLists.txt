cmake_minimum_required(VERSION 3.22.1)
project(wa_legacy C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CORE "${ROOT}/core")
set(DEPS "${ROOT}/native-deps/${ANDROID_ABI}")
if(NOT EXISTS "${DEPS}/lib/libcrypto.a")
    message(FATAL_ERROR "Run scripts/build-native-deps.sh first")
endif()
add_library(crypto STATIC IMPORTED)
set_target_properties(crypto PROPERTIES IMPORTED_LOCATION "${DEPS}/lib/libcrypto.a")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/native-deps/signal" "${CMAKE_CURRENT_BINARY_DIR}/signal")
set_target_properties(signal-protocol-c PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(wa_legacy SHARED bridge.c "${CORE}/noise.c" "${CORE}/wire.c" "${CORE}/ws.c" "${CORE}/handshake.c" "${CORE}/register.c" "${CORE}/identity.c" "${CORE}/node.c" "${CORE}/pairing.c" "${CORE}/messaging.c" "${CORE}/inbox.c" "${CORE}/history.c" "${CORE}/signal_engine.c" "${CORE}/crypto_signal.c" "${CORE}/storage_posix.c")
target_include_directories(wa_legacy PRIVATE "${CORE}" "${ROOT}/native-deps/signal/src" "${DEPS}/include")
target_compile_options(wa_legacy PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
target_link_libraries(wa_legacy PRIVATE signal-protocol-c crypto dl z)
